/* ===== Gemini AI — Updated model ===== */ const G = 'https://generativelanguage.googleapis.com/v1beta/'; async function findModel(key) { const r = await fetch(G + 'models?pageSize=100', { headers: { 'x-goog-api-key': key } }); if (!r.ok) { throw new Error( 'Unable to list Gemini models. Check your API key and internet connection.' ); } const data = await r.json(); const models = (data.models || []).filter(m => (m.supportedGenerationMethods || []).includes('generateContent') ); const preferred = [ 'gemini-3.8-flash', 'gemini-3.6-flash', 'gemini-3.5-flash-lite', 'gemini-flash-latest' ]; const model = preferred.find(name => models.some(m => m.name === 'models/' + name) ); if (!model) { throw new Error( 'No supported Gemini Flash model is available for this API key.' ); } localStorage.setItem('zpi_model', model); return model; } async function gemini(text) { let key = await loadKey('gemini'); if (!key) { if (!(await setKey('gemini', 'Gemini'))) { throw new Error( 'No AI key saved. Tap the AI key button to add one.' ); } key = await loadKey('gemini'); } // Do not reuse the old, unavailable model name. let model = 'gemini-3.8-flash'; localStorage.setItem('zpi_model', model); for (let attempt = 0; attempt < 2; attempt++) { let r; try { r = await fetch( G + 'models/' + model + ':generateContent', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': key }, body: JSON.stringify({ systemInstruction: { parts: [{ text: 'You are ZPI, a helpful voice assistant. Answer briefly in the same language the user speaks.' }] }, contents: [{ role: 'user', parts: [{ text }] }] }) } ); } catch { throw new Error( 'Could not connect to Gemini. Check your internet connection.' ); } if (r.ok) { const data = await r.json(); const answer = ( data.candidates?.[0]?.content?.parts || [] ).map(part => part.text || '').join('').trim(); if (!answer) { throw new Error( 'Gemini returned an empty answer. Please try again.' ); } localStorage.setItem('zpi_model', model); return answer; } const data = await r.json().catch(() => ({})); const message = data.error?.message || ''; if (r.status === 400 || r.status === 403) { throw new Error( 'Gemini API key or request rejected. Check your API key and model access.' ); } if (r.status === 429) { throw new Error( 'Gemini API limit reached. Please try again later.' ); } if ( attempt === 0 && (r.status === 404 || /not found|no longer available|unsupported model/i.test(message)) ) { model = await findModel(key); continue; } throw new Error( 'Gemini API error: ' + (message || r.status) ); } throw new Error('Unable to get an AI response. Please try again.'); }
